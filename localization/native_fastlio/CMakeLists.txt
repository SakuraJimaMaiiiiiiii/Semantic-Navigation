cmake_minimum_required(VERSION 3.20)
project(native_fastlio LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(PYBIND11_FINDPYTHON ON)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
find_package(Eigen3 CONFIG REQUIRED)
find_package(Boost REQUIRED)
find_package(Threads REQUIRED)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/generated/IMU_Processing.hpp")
  message(FATAL_ERROR "Run python localization/native_fastlio/prepare_source.py first")
endif()
pybind11_add_module(_native_fastlio NO_EXTRAS bindings.cpp generated/ikd_Tree.cpp)
target_include_directories(_native_fastlio PRIVATE . generated ${Boost_INCLUDE_DIRS})
target_link_libraries(_native_fastlio PRIVATE Eigen3::Eigen Threads::Threads)
target_compile_definitions(_native_fastlio PRIVATE _USE_MATH_DEFINES NOMINMAX)
if(MSVC)
  target_compile_options(_native_fastlio PRIVATE /bigobj /EHsc /O2)
endif()
set_target_properties(_native_fastlio PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/.." RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/..")
foreach(CONFIG DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
 set_target_properties(_native_fastlio PROPERTIES LIBRARY_OUTPUT_DIRECTORY_${CONFIG} "${CMAKE_CURRENT_SOURCE_DIR}/.." RUNTIME_OUTPUT_DIRECTORY_${CONFIG} "${CMAKE_CURRENT_SOURCE_DIR}/..")
endforeach()
